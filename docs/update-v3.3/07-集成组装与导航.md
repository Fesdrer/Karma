### 1.5 XiaoLiuRenContent 组合组件

> 此组件可以放在 DivinationScreen.kt 文件内（作为 private composable），也可以放在独立的 `components/` 文件中。建议放在 DivinationScreen.kt 中以减少文件碎片。

```kotlin
@Composable
private fun XiaoLiuRenContent(
    state: DivinationUiState,
    onInputModeChanged: (InputMode) -> Unit,
    onMonthChanged: (Int) -> Unit,
    onDayChanged: (Int) -> Unit,
    onShiChenChanged: (ShiChen) -> Unit,
    onNumber1Changed: (String) -> Unit,
    onNumber2Changed: (String) -> Unit,
    onNumber3Changed: (String) -> Unit,
    onStartClick: () -> Unit,
    onPhaseComplete: (AnimationPhase) -> Unit,
    onRetry: () -> Unit,
) {
    val palacePositions = remember { mutableStateListOf<PalacePosition>() }

    Box(modifier = Modifier.fillMaxSize().background(Color.Black)) {

        // 层 1：三柱六宫 Canvas（始终显示）
        XiaoLiuRenPillarCanvas(
            highlightedIndex = if (state.fullPath.isNotEmpty() && !state.isAnimating)
                state.fullPath.last() else null,
            resultIndex = if (state.animationPhase == AnimationPhase.RESULT_GLOW ||
                              state.animationPhase == AnimationPhase.COMPLETE)
                state.resultPalace?.index?.index else null,
            onPalacePositionsReady = { positions ->
                palacePositions.clear()
                palacePositions.addAll(positions)
            },
            modifier = Modifier.fillMaxSize(),
        )

        // 层 2：金线动画 Canvas（动画阶段显示）
        if (state.isAnimating && state.fullPath.isNotEmpty() && palacePositions.size == 6) {
            XiaoLiuRenThreadCanvas(
                fullPath = state.fullPath,
                palacePositions = palacePositions.toList(),
                animationPhase = state.animationPhase,
                onPhaseComplete = onPhaseComplete,
                modifier = Modifier.fillMaxSize(),
            )
        }

        // 层 3：输入面板（IDLE 阶段显示）
        if (state.animationPhase == AnimationPhase.IDLE) {
            XiaoLiuRenInputPanel(
                inputMode = state.inputMode,
                month = state.month,
                day = state.day,
                shiChen = state.shiChen,
                number1 = state.number1,
                number2 = state.number2,
                number3 = state.number3,
                onInputModeChanged = onInputModeChanged,
                onMonthChanged = onMonthChanged,
                onDayChanged = onDayChanged,
                onShiChenChanged = onShiChenChanged,
                onNumber1Changed = onNumber1Changed,
                onNumber2Changed = onNumber2Changed,
                onNumber3Changed = onNumber3Changed,
                onStartClick = onStartClick,
                enabled = !state.isAnimating,
                modifier = Modifier.fillMaxSize(),
            )
        }

        // 层 4：结果面板（COMPLETE 阶段显示）
        if (state.animationPhase == AnimationPhase.COMPLETE && state.resultPalace != null) {
            XiaoLiuRenResultPanel(
                result = state.resultPalace!!,
                onRetry = onRetry,
                modifier = Modifier.fillMaxSize(),
            )
        }
    }
}
```

### 1.6 状态与 UI 对应关系

| animationPhase | 柱 Canvas | 金线 Canvas | 输入面板 | 结果面板 |
|:---|:---:|:---:|:---:|:---:|
| IDLE | ✅ | ❌ | ✅ | ❌ |
| COUNTING_MONTH | ✅ | ✅ | ❌ | ❌ |
| MONTH_PAUSE | ✅ | ✅ | ❌ | ❌ |
| COUNTING_DAY | ✅ | ✅ | ❌ | ❌ |
| DAY_PAUSE | ✅ | ✅ | ❌ | ❌ |
| COUNTING_HOUR | ✅ | ✅ | ❌ | ❌ |
| RESULT_GLOW | ✅ | ✅（发光）| ❌ | ❌ |
| COMPLETE | ✅ | ❌ | ❌ | ✅ |

### 1.7 PalacePosition 回调处理

`XiaoLiuRenPillarCanvas` 通过 `onPalacePositionsReady` 回调传递六宫坐标。由于 Canvas 的坐标在首次绘制时确定，此回调在 Canvas 的 `DrawScope` 内被调用。

为避免在 draw 阶段修改状态导致的问题，使用 `LaunchedEffect` 延迟一帧处理：

```kotlin
var tempPositions by remember { mutableStateOf<List<PalacePosition>?>(null) }

LaunchedEffect(tempPositions) {
    tempPositions?.let { positions ->
        palacePositions.clear()
        palacePositions.addAll(positions)
    }
}

// PillarCanvas 中：
// onPalacePositionsReady 回调在 layout 阶段后调用，触发 recomposition
```
