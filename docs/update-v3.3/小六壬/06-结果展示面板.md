@Composable
private fun PalaceNameCard(result: PalaceRevelation) {
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(Color(0xFF16213e))  // PanelBg
            .border(1.dp, Color(0xFF334444), RoundedCornerShape(12.dp))
            .padding(16.dp),
        contentAlignment = Alignment.Center,
    ) {
        Column(horizontalAlignment = Alignment.CenterHorizontally) {
            // 宫名
            Text(
                text = result.name,
                fontSize = 42.sp,
                fontWeight = FontWeight.Bold,
                color = Color(0xFFffd700),  // Gold
            )
            // 吉凶徽章
            Spacer(Modifier.height(6.dp))
            Box(
                modifier = Modifier
                    .clip(RoundedCornerShape(6.dp))
                    .background(
                        if (result.fortuneLevel.isAuspicious)
                            Color(0xFF69f0ae).copy(alpha = 0.2f)  // GreenPositive tint
                        else
                            Color(0xFFff5252).copy(alpha = 0.2f)  // RedNegative tint
                    )
                    .padding(horizontal = 12.dp, vertical = 4.dp),
            ) {
                Text(
                    text = result.fortuneLevel.label,
                    color = if (result.fortuneLevel.isAuspicious)
                        Color(0xFF69f0ae) else Color(0xFFff5252),
                    fontSize = 16.sp,
                    fontWeight = FontWeight.Bold,
                )
            }
            // 五行 · 方位 · 六神
            Spacer(Modifier.height(8.dp))
            Text(
                text = "${result.wuxing} · ${result.direction} · ${result.sixGods}",
                fontSize = 14.sp,
                color = Color(0xFFa0c4ff),  // BlueLight
            )
        }
    }
}
