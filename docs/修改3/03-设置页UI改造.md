`ui/settings/SettingsScreen.kt`

## 具体代码

将整个 `EventSettingsCard` 函数替换为：

```kotlin
@Composable
private fun EventSettingsCard(
    draft: com.example.karma.data.local.entity.KarmaSettingsEntity,
    viewModel: SettingsViewModel,
) {
    SettingsCard("右边事件列表") {
        Text(
            "每行一个事件，直接编辑即可增删",
            fontSize = 12.sp,
            color = TextSecondary,
        )
        Spacer(Modifier.height(12.dp))

        // — 善业 —
        EventSection(
            title = "善业",
            titleColor = Color(0xFF69f0ae), // 绿色
            events = draft.goodDeedPresets,
            onUpdate = { viewModel.updateGoodDeedPresets(it) },
        )

        Spacer(Modifier.height(12.dp))

        // — 恶业 —
        EventSection(
            title = "恶业",
            titleColor = Color(0xFFff5252), // 红色
            events = draft.badDeedPresets,
            onUpdate = { viewModel.updateBadDeedPresets(it) },
        )

        Spacer(Modifier.height(12.dp))

        // — 善果 —
        EventSection(
            title = "善果",
            titleColor = Color(0xFFffd700), // 金色
            events = draft.goodResultPresets,
            onUpdate = { viewModel.updateGoodResultPresets(it) },
        )
    }
}

@Composable
private fun EventSection(
    title: String,
    titleColor: Color,
    events: List<String>,
    onUpdate: (String) -> Unit,
) {
    var text by remember(events) {
        mutableStateOf(events.joinToString("\n"))
    }

    Column {
        Text(
            text = "■ $title",
            fontSize = 14.sp,
            fontWeight = FontWeight.Bold,
            color = titleColor,
        )
        Spacer(Modifier.height(4.dp))
        OutlinedTextField(
            value = text,
            onValueChange = { newText ->
                text = newText
                onUpdate(newText)
            },
            modifier = Modifier
                .fillMaxWidth()
                .heightIn(min = 80.dp),
            singleLine = false,
            minLines = 3,
            textStyle = MaterialTheme.typography.bodySmall,
            placeholder = { Text("输入$title事件...") },
            colors = OutlinedTextFieldDefaults.colors(
                focusedBorderColor = titleColor,
                unfocusedBorderColor = BorderSubtle,
                focusedTextColor = TextPrimary,
                unfocusedTextColor = TextPrimary,
                cursorColor = titleColor,
            ),
        )
    }
}
```

## 关键点

- 每个文本框有自己的颜色主题（善业绿、恶业红、善果金），focusedBorderColor 和 title 使用对应色
- 使用 `remember(events)` 确保事件列表变化时文本框内容同步更新
- 每个文本框独立编辑，互不影响
- 提取 `EventSection` 辅助 composable 避免重复代码
