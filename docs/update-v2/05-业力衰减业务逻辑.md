### applyDecay() 方法

在 `KarmaRepository` 类内新增：

```kotlin
/**
 * 应用业力衰减。
 * 在应用启动时调用，检查自上次扣除以来漏扣的天数，一次性补扣。
 * 
 * @return 本次扣除的总分数（0 表示未扣除）
 */
suspend fun applyDecay(): Float {
    val settings = settingsDao.getSettingsOnce() ?: KarmaSettingsEntity()
    
    // 衰减未启用 → 跳过
    if (!settings.decayEnabled) return 0f
    
    val today = LocalDate.now()
    val todayStr = today.format(DateTimeFormatter.ISO_LOCAL_DATE)
    
    // 今天已经扣过 → 跳过
    if (settings.lastDecayDate == todayStr) return 0f
    
    // 确定上次扣除日期
    val lastDate: LocalDate = if (settings.lastDecayDate.isEmpty()) {
        // 从未扣除过：从第一条历史记录的次日开始
        val firstEntry = historyDao.getOldestEntry()
        if (firstEntry == null) {
            // 没有历史记录 → 从今天开始
            settingsDao.updateLastDecayDate(todayStr)
            return 0f
        }
        // 首条记录的日期
        val firstDate = timestampToLocalDate(firstEntry.timestamp)
        firstDate.minusDays(1)  // 从首条记录的次日开始扣
    } else {
        LocalDate.parse(settings.lastDecayDate, DateTimeFormatter.ISO_LOCAL_DATE)
    }
    
    val daysToCatchUp = ChronoUnit.DAYS.between(lastDate, today).toInt()
    if (daysToCatchUp <= 0) return 0f
    
    // 逐天补扣
    var totalDeducted = 0f
    var currentScore = settings.totalScore
    val amounts = settings.rankDecayAmounts
    
    for (dayOffset in 1..daysToCatchUp) {
        val decayDate = lastDate.plusDays(dayOffset.toLong())
        val rank = getRankForScore(currentScore)
        val deduction = getDecayAmount(rank, amounts)
        
        if (deduction > 0f) {
            currentScore = roundToOneDecimal(currentScore - deduction)
            totalDeducted += deduction
            
            val timestampMs = decayDate
                .atStartOfDay(ZoneId.systemDefault())
                .toInstant()
                .toEpochMilli()
            
            historyDao.insertEntry(
                HistoryEntryEntity(
                    timestamp = timestampMs,
                    delta = -deduction,
                    event = "业力衰减",
                    type = "decay",
                    totalAfter = currentScore,
                )
            )
        }
    }
    
    // 更新数据库
    val updatedSettings = settings.copy(
        totalScore = currentScore,
        lastDecayDate = todayStr,
    )
    settingsDao.upsertSettings(updatedSettings)
    
    return totalDeducted
}

// ===== 辅助方法 =====

/**
 * 根据总分确定阶位序号（1-based）。
 * 分数 < 0 视为壹阶（index=0）。
 */
private fun getRankForScore(score: Float): Int {
    return when {
        score < 0f -> 0
        score < 10f -> 0    // 壹阶
        score < 30f -> 1    // 贰阶
        score < 60f -> 2    // 叁阶
        score < 100f -> 3   // 肆阶
        score < 150f -> 4   // 伍阶
        score < 210f -> 5   // 陆阶
        score < 280f -> 6   // 柒阶
        score < 360f -> 7   // 捌阶
        else -> 8           // 玖阶
    }
}

/**
 * 获取指定阶位的衰减扣除量。
 * @param rank 阶位索引（0-based，0=壹阶）
 * @param amounts 9个阶位的扣除量列表
 */
private fun getDecayAmount(rank: Int, amounts: List<Float>): Float {
    if (amounts.isEmpty()) return 2f
    val index = rank.coerceIn(0, amounts.size - 1)
    return amounts[index]
}

/**
 * 将时间戳转为 LocalDate。
 */
private fun timestampToLocalDate(timestamp: Long): LocalDate {
    return java.time.Instant.ofEpochMilli(timestamp)
        .atZone(ZoneId.systemDefault())
        .toLocalDate()
}
```

### 关于 roundToOneDecimal

该方法已在 Repository 中存在（`private fun roundToOneDecimal(value: Float): Float`），直接使用。

### 关于 ranking 重叠

当前 `Rank.RANKS` 定义（在 Rank.kt）：
- 壹阶: 0~10, 贰阶: 10~30, 叁阶: 30~60, ...

`getRankForScore()` 使用左闭右开区间匹配 `score < max` 逻辑，与 `Rank.RANKS.find { score >= it.min && score < it.max }` 一致。

当分数为负数时视为壹阶（扣除量取 `amounts[0]`），防止 index out of bounds。

### 关于历史记录时间戳

每天衰减记录的时间戳设为当天 `00:00:00`（用户所在时区），确保在历史图表中落在正确日期。这和使用 `System.currentTimeMillis()` 的记录有所不同（后者是精确时间点），但在日/周/月视图中均会落在该日。

## 5.3 KarmaApplication.onCreate 中调用

**文件**: `app/src/main/java/com/example/karma/data/KarmaApplication.kt`

### 改动

确保在 Application 启动时也触发衰减检查，应对从通知栏或桌面快捷方式等非 MainScreen 入口进入的情况。

```kotlin
package com.example.karma

import android.app.Application
import com.example.karma.di.AppContainer
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.launch

class KarmaApplication : Application() {

    lateinit var appContainer: AppContainer
        private set

    private val applicationScope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    override fun onCreate() {
        super.onCreate()
        appContainer = AppContainer(this)

        // ★ 启动时触发衰减检查
        applicationScope.launch {
            try {
                val deducted = appContainer.repository.applyDecay()
                if (deducted > 0f) {
                    android.util.Log.i("KarmaDecay", "衰减扣除: $deducted 分")
                }
            } catch (e: Exception) {
                android.util.Log.w("KarmaDecay", "衰减检查异常", e)
            }
        }
    }
}
```

**注意**: 
- 使用 `Dispatchers.IO` 避免阻塞主线程
- 使用 `SupervisorJob()` 确保异常不会传播到全局协程作用域
- Application 中不能直接使用 `viewModelScope`，所以使用自定义 `applicationScope`
- 日志输出方便调试
